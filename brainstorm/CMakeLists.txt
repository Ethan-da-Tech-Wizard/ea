cmake_minimum_required(VERSION 3.21)
project(archi_brainstorm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
set(ARCHI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../archi")
set(ARCHI_HEADERS "${ARCHI_SOURCE}/src/DocumentFiles.h" "${ARCHI_SOURCE}/src/MathEngine.h" "${ARCHI_SOURCE}/src/DocumentSync.h")
add_executable(archi-board board.cpp ${ARCHI_HEADERS})
target_compile_definitions(archi-board PRIVATE ARCHI_MATH_PYTHON="${ARCHI_SOURCE}/.venv-math/bin/python" ARCHI_MATH_SCRIPT="${ARCHI_SOURCE}/src/math_engine.py")
target_link_libraries(archi-board PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
option(BOARD_TESTS "Build board interaction tests" OFF)
if(BOARD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS QuickTest Test)
    add_executable(board-tests tests/setup.cpp ${ARCHI_HEADERS})
    target_link_libraries(board-tests PRIVATE Qt6::QuickTest Qt6::QuickControls2)
    target_compile_definitions(board-tests PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
    target_compile_definitions(board-tests PRIVATE ARCHI_MATH_PYTHON="${ARCHI_SOURCE}/.venv-math/bin/python" ARCHI_MATH_SCRIPT="${ARCHI_SOURCE}/src/math_engine.py")
    add_executable(sync-tests tests/test_sync.cpp "${ARCHI_SOURCE}/src/DocumentSync.h")
    target_link_libraries(sync-tests PRIVATE Qt6::Core Qt6::Test)
endif()
